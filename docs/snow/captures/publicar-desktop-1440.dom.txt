	function publicar_parte(){
		var form = document.formulario;
		form_reiniciar(form);

		if(form.nombre.value.length < 4)
			form_add_error(form.nombre,'Debes rellenar tu nombre');
		if(!ValidarEmail(form.email.value))
			form_add_error(form.email,'Debes introducir un email válido');
		if(parseInt(form.estacion[form.estacion.selectedIndex].value) == 0)
			form_add_error(form.estacion,'Debes seleccionar la estación');
		if(form.texto.value.length < 10)
			form_add_error(form.texto,'Debes comentar el estado de la estación');

		if(!$(form).find("#formerrors").data("error")) form.submit();
	}
